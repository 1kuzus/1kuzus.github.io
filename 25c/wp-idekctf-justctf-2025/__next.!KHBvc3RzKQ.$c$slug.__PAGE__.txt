    payload = b""
    for offset, ch in enumerate(s):
        offset = bytes([offset])

            ch_xor_0xff = bytes([ord(ch) ^ 0xFF])

            payload += b"\x02\x00"  # pop(R0)
            payload += b"\x01\xff"  # push(0xFF)
            payload += b"\x02\x07"  # pop(R7)
            payload += b"\x05\x07"  # R0 ^= R7
